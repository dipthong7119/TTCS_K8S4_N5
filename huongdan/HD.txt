cách để push lên git
git config --global user.name "tên đăng nhập"
git config --global user.email "mail của bản thân"
cd vào thư mục chứa code
git switch <tên nhánh tên của bạn thân + họ>
git branch #kiểm tra nhánh
git add .
git commit -m "viết commit vào đây"
git push origin tên nhánh
cách clone về 
git clone -b tên nhánh https://github.com/dipthong7119/TTCS_K8S4_N5.
